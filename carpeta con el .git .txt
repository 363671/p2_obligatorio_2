using System;
using System.IO;
using System.Linq;
using System.Reflection;

// Cambia la ruta a la carpeta bin\Debug\... de tu proyecto
string rutaDll = @"C:\Users\diegu\OneDrive - Facultad de Ingenieria - Universidad ORT Uruguay\P2\2 Obligatorio parte 2\Obligatorio_P2_Fiscalia\Dominio\bin\Debug\net10.0\Dominio.dll";
var ensamblado = Assembly.LoadFrom(rutaDll);

using (var writer = new StreamWriter(@"C:\boot\metodos.txt"))

foreach (var tipo in ensamblado.GetTypes().OrderBy(t => t.FullName))
{
    writer.WriteLine($"Clase: {tipo.FullName}");
    var metodos = tipo.GetMethods(BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance | BindingFlags.Static | BindingFlags.DeclaredOnly);
    
    foreach (var m in metodos)
    {
        writer.WriteLine($"   - {m.ReturnType.Name} {m.Name}()");
    }
}
